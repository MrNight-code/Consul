### 1. Obtener Contadores (GET)

**Curl:**
```bash

  -H "Authorization: Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..." \
  -H "X-Tenant-Id: foret" \
  -H "Content-Type: application/json"
```

**PowerShell:**
```powershell
$headers = @{
    "Authorization" = "Bearer $token"
    "X-Tenant-Id" = "foret"
}

$response = Invoke-RestMethod `
    -Uri "http://localhost:5000/api/Dashboard/1" `
    -Headers $headers `
    -Method Get

$response | ConvertTo-Json -Depth 10
```

**JavaScript/Fetch:**

### 3. Flujo Completo (Frontend)

```typescript
// 1. Obtener contadores iniciales
async function cargarDashboard(condominioId: number) {
  try {
    const response = await fetch(`/api/Dashboard/${condominioId}`, {
      headers: {
        'Authorization': `Bearer ${localStorage.getItem('token')}`,
        'X-Tenant-Id': localStorage.getItem('tenant')
      }
    });
    
    if (!response.ok) throw new Error('Error al cargar dashboard');
    
    const data = await response.json();
    actualizarUI(data);
  } catch (error) {
    mostrarError(error);
  }
}

// 2. Botón de Refrescar
async function refrescarDashboard(condominioId: number) {
  try {
    mostrarSpinner();
    
    const response = await fetch(`/api/Dashboard/${condominioId}/refrescar`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${localStorage.getItem('token')}`,
        'X-Tenant-Id': localStorage.getItem('tenant')
      }
    });
    
    if (!response.ok) throw new Error('Error al refrescar');
    
    const data = await response.json();
    actualizarUI(data);
    mostrarNotificacion('Contadores actualizados');
  } catch (error) {
    mostrarError(error);
  } finally {
    ocultarSpinner();
  }
}

// 3. Actualizar UI
function actualizarUI(data: DashboardCountersDto) {
  document.getElementById('totalUnidades').textContent = data.totalUnidades;
  document.getElementById('unidadesEnMora').textContent = data.unidadesEnMora;
  document.getElementById('totalCobrado').textContent = 
    `$${data.totalCobradoMesActual.toFixed(2)}`;
  document.getElementById('porcentajeCobranza').textContent = 
    `${data.porcentajeCobranza.toFixed(2)}%`;
  document.getElementById('ultimaActualizacion').textContent = 
    new Date(data.ultimaActualizacion).toLocaleString();
}
```

```typescript
async function obtenerContadores(condominioId: number) {

    if (response.status === 404) {
      // Condominio no encontrado
      console.error('El condominio no existe');
      return null;
    }
    
    if (response.status === 400) {
      const error = await response.json();
      console.error('Error en cálculo:', error.message);
      return null;
    }
    
    if (!response.ok) {
      throw new Error(`HTTP error! status: ${response.status}`);
    }
    
    return await response.json();
  } catch (error) {
    console.error('Error:', error);
    return null;
  }
}
```

### 5. Integración con Angular

```typescript
import { Injectable } from '@angular/core';
import { HttpClient, HttpHeaders } from '@angular/common/http';
import { Observable } from 'rxjs';

@Injectable({ providedIn: 'root' })
export class DashboardService {
  private baseUrl = 'http://localhost:5000/api/Dashboard';

  constructor(private http: HttpClient, private authService: AuthService) {}

  getContadores(condominioId: number): Observable<DashboardCountersDto> {
    const headers = new HttpHeaders({
      'Authorization': `Bearer ${this.authService.getToken()}`,
      'X-Tenant-Id': this.authService.getTenant()
    });

    return this.http.get<DashboardCountersDto>(
      `${this.baseUrl}/${condominioId}`,
      { headers }
    );
  }

  refrescarContadores(condominioId: number): Observable<DashboardCountersDto> {
    const headers = new HttpHeaders({
      'Authorization': `Bearer ${this.authService.getToken()}`,
      'X-Tenant-Id': this.authService.getTenant()
    });

    return this.http.post<DashboardCountersDto>(
      `${this.baseUrl}/${condominioId}/refrescar`,
      {},
      { headers }
    );
  }
}

// Uso en Component
@Component({
  selector: 'app-dashboard',
  template: `
    <div *ngIf="contadores$ | async as contadores">
      <p>Total Unidades: {{contadores.totalUnidades}}</p>
      <p>En Mora: {{contadores.unidadesEnMora}}</p>
      <p>Cobrado: ${{contadores.totalCobradoMesActual}}</p>
      <p>Cobranza: {{contadores.porcentajeCobranza}}%</p>
      <button (click)="refrescar()">Refrescar</button>
    </div>
  `
})
export class DashboardComponent implements OnInit {
  contadores$: Observable<DashboardCountersDto>;
  condominioId = 1;

  constructor(private dashboardService: DashboardService) {}

  ngOnInit() {
    this.cargarContadores();
  }

  cargarContadores() {
    this.contadores$ = this.dashboardService.getContadores(this.condominioId);
  }

  refrescar() {
    this.contadores$ = this.dashboardService.refrescarContadores(this.condominioId);
  }
}
```
